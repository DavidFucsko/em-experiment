# PowerShell, run from em-experiment/. Cohorts: k1, ked ($env:EM_COHORT). Scenarios (scenarios.py): full, core_id,
# initials, no_names, safe_harbor, sex_race_age (default: all, or e.g. --scenario no_names safe_harbor). Pools: natural, hard.
# Every model run saves, one row per candidate pair, its scores (comparison vector, match probability; m and u in the
# .json) to data/matches_<cohort>/<model>_<scenario>_<pool>.scores.parquet and its links to ....links.parquet;
# compare.py and explain_weights.py read stored runs, so a model must have run before it is compared.

# ===== ALL IN ONE: data, every model (D, D+C, D+E-stat, D+E-ktx), the comparisons, the fastLink check (needs R with fastLink) =====
# K1
$start = Get-Date; "Start: $start"; $env:EM_COHORT = "k1"; uv run python -m src.preprocessing.temporal_split; uv run python -m src.preprocessing.code_titles; uv run python -m src.preprocessing.identity_assignment; uv run python -m src.preprocessing.code_vectors; foreach ($pool in "natural","hard") { uv run python -u -m src.experiments.demographic_baseline --pool $pool | tee results/k1_D_${pool}.txt; uv run python -u -m src.experiments.clinical_exact --pool $pool | tee results/k1_D+C_${pool}.txt; foreach ($vectors in "stat","ktx") { uv run python -u -m src.experiments.clinical_embedded --vectors $vectors --pool $pool | tee results/k1_D+E-${vectors}_${pool}.txt }; foreach ($pair in "D,D+C","D+C,D+E-stat","D+C,D+E-ktx") { $base, $new = $pair -split ","; uv run python -u -m src.experiments.compare --base $base --new $new --pool $pool | tee results/k1_compare_${base}_vs_${new}_${pool}.txt }; uv run python -u -m src.experiments.link_review_or_new --pool $pool | tee results/k1_link_review_or_new_${pool}.txt }; uv run python -u -m src.experiments.demographic_baseline_fastlink | tee results/k1_fastlink_natural.txt; "End: $(Get-Date)   Duration: $((Get-Date) - $start)"
# K-ED (the cohort-size axis): the preprocessing on the whole cohort, then every run on nested random subsets of 12,275 (= K1's size: K1 vs K-ED at equal size is the cohort effect), 18,000 and 24,000 patients ($env:EM_SIZE), natural pool only. The four models run at the same time as background jobs (each writes its own log in results/), the comparisons after all four have finished
$start = Get-Date; "Start: $start"; $env:EM_COHORT = "ked"; $env:EM_SIZE = ""; uv run python -m src.preprocessing.temporal_split; uv run python -m src.preprocessing.code_titles; uv run python -m src.preprocessing.identity_assignment; uv run python -m src.preprocessing.code_vectors; $dir = (Get-Location).Path; $model = { param($dir, $size, $step, $log) Set-Location $dir; $env:EM_COHORT = "ked"; $env:EM_SIZE = $size; Invoke-Expression "uv run python -u -m src.experiments.$step" *> $log }; foreach ($size in "12275","18000","24000") { $env:EM_SIZE = $size; $c = "ked_$size"; foreach ($pool in "natural") { $runs = @(@("demographic_baseline --pool $pool", "D"), @("clinical_exact --pool $pool", "D+C"), @("clinical_embedded --vectors stat --pool $pool", "D+E-stat"), @("clinical_embedded --vectors ktx --pool $pool", "D+E-ktx")); $logs = $runs | ForEach-Object { "$dir/results/${c}_$($_[1])_${pool}.txt" }; $jobs = for ($r = 0; $r -lt $runs.Count; $r++) { Start-Job -ScriptBlock $model -ArgumentList $dir, $size, $runs[$r][0], $logs[$r] }; $jobs | Wait-Job | Receive-Job; $jobs | Remove-Job; Get-Content $logs; foreach ($pair in "D,D+C","D+C,D+E-stat","D+C,D+E-ktx") { $base, $new = $pair -split ","; uv run python -u -m src.experiments.compare --base $base --new $new --pool $pool | tee results/${c}_compare_${base}_vs_${new}_${pool}.txt }; uv run python -u -m src.experiments.link_review_or_new --pool $pool | tee results/${c}_link_review_or_new_${pool}.txt }; uv run python -u -m src.experiments.demographic_baseline_fastlink | tee results/${c}_fastlink_natural.txt }; $env:EM_SIZE = ""; $env:EM_COHORT = "k1"; "End: $(Get-Date)   Duration: $((Get-Date) - $start)"

# ===== NEW SCENARIOS ON K1 ONLY: the same as K1's ALL IN ONE without the preprocessing (its data is current), for the scenarios added after it (core_id, initials, safe_harbor, sex_race_age); logs get _scenarios, so the full runs' logs are kept =====
$start = Get-Date; "Start: $start"; $env:EM_COHORT = "k1"; $env:EM_SIZE = ""; $sc = "core_id","initials","safe_harbor","sex_race_age"; foreach ($pool in "natural","hard") { uv run python -u -m src.experiments.demographic_baseline --scenario $sc --pool $pool | tee results/k1_scenarios_D_${pool}.txt; uv run python -u -m src.experiments.clinical_exact --scenario $sc --pool $pool | tee results/k1_scenarios_D+C_${pool}.txt; foreach ($vectors in "stat","ktx") { uv run python -u -m src.experiments.clinical_embedded --vectors $vectors --scenario $sc --pool $pool | tee results/k1_scenarios_D+E-${vectors}_${pool}.txt }; foreach ($pair in "D,D+C","D+C,D+E-stat","D+C,D+E-ktx") { $base, $new = $pair -split ","; uv run python -u -m src.experiments.compare --base $base --new $new --scenario $sc --pool $pool | tee results/k1_scenarios_compare_${base}_vs_${new}_${pool}.txt }; uv run python -u -m src.experiments.link_review_or_new --scenario $sc --pool $pool | tee results/k1_scenarios_link_review_or_new_${pool}.txt }; uv run python -u -m src.experiments.demographic_baseline_fastlink --scenario $sc | tee results/k1_scenarios_fastlink_natural.txt; "End: $(Get-Date)   Duration: $((Get-Date) - $start)"
# and their explanations (after the line above)
$env:EM_COHORT = "k1"; $env:EM_SIZE = ""; $sc = "core_id","initials","safe_harbor","sex_race_age"; foreach ($pool in "natural","hard") { foreach ($m in "D","D+C","D+E-stat","D+E-ktx") { uv run python -u -m src.experiments.explain_weights --model $m --scenario $sc --pool $pool | tee results/k1_scenarios_explain_${m}_${pool}.txt } }

# ===== EXPLAIN ALL: weights, contributions and the decisions changed against D for every stored run (K1 both pools, K-ED's three sizes natural); the models must have run first =====
$start = Get-Date; $models = "D","D+C","D+E-stat","D+E-ktx"; $env:EM_COHORT = "k1"; $env:EM_SIZE = ""; foreach ($pool in "natural","hard") { foreach ($m in $models) { uv run python -u -m src.experiments.explain_weights --model $m --pool $pool | tee results/k1_explain_${m}_${pool}.txt } }; $env:EM_COHORT = "ked"; foreach ($size in "12275","18000","24000") { $env:EM_SIZE = $size; foreach ($m in $models) { uv run python -u -m src.experiments.explain_weights --model $m --pool natural | tee results/ked_${size}_explain_${m}_natural.txt } }; $env:EM_SIZE = ""; $env:EM_COHORT = "k1"; "End: $(Get-Date)   Duration: $((Get-Date) - $start)"

# ===== PRIORS of m and u (option; priors.PRIORS): winkler (default: δ = 0.001, Winkler 2002) | none (maximum likelihood) | flat | centred. Run names get -none / -flat / -centred, so all three are kept; compare against the default on the same candidate pairs =====
$env:EM_COHORT = "k1"; $env:EM_SIZE = ""; foreach ($prior in "none","flat","centred") { uv run python -u -m src.experiments.demographic_baseline --pool natural --prior $prior | tee results/k1_D-${prior}_natural.txt; uv run python -u -m src.experiments.clinical_exact --pool natural --prior $prior | tee results/k1_D+C-${prior}_natural.txt; foreach ($m in "D","D+C") { uv run python -u -m src.experiments.compare --base $m --new "$m-$prior" --pool natural | tee results/k1_compare_${m}_vs_${m}-${prior}_natural.txt } }

# ===== SINGLE STEPS (examples; the cohort is $env:EM_COHORT, default k1) =====
uv run python -m src.experiments.demographic_baseline --scenario no_names --pool natural
uv run python -m src.experiments.clinical_exact --scenario no_names --pool natural
uv run python -m src.experiments.clinical_embedded --vectors stat --scenario no_names --pool natural
uv run python -m src.experiments.clinical_embedded --vectors ktx --scenario no_names --pool natural
uv run python -m src.experiments.compare --base D --new D+C --scenario no_names --pool natural
uv run python -m src.experiments.compare --base D+C --new D+E-stat --scenario no_names --pool natural

# independent clinical fields (naive), the comparison for the composite field
uv run python -m src.experiments.clinical_exact --variant naive --scenario no_names --pool natural
uv run python -m src.experiments.clinical_embedded --vectors stat --variant naive --scenario no_names --pool natural
uv run python -m src.experiments.compare --base D+C-naive --new D+C --scenario no_names --pool natural

# how the stored models would run in a master patient index (automatic link / review / new record per incoming record)
uv run python -m src.experiments.link_review_or_new --pool natural

# what a stored run weighs and why it links
uv run python -m src.experiments.explain_weights --model D+C --scenario no_names --pool natural
